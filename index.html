<script>

    let input = "";

    let studentData = {}; // 번호:이름 매핑 저장용

    const display = document.getElementById('display');

    const status = document.getElementById('status');

    const studentNameDiv = document.getElementById('studentName');

    const spinner = document.getElementById('spinner');


    // --- Coda API 설정 ---

    const API_TOKEN = "YOUR_CODA_API_TOKEN";

    const DOC_ID = "YOUR_DOC_ID";

    const STUDENT_TABLE_ID = "STUDENT_LIST_TABLE_ID"; // 학생 명단 테이블

    const LOG_TABLE_ID = "ATTENDANCE_LOG_TABLE_ID";   // 출결 기록용 테이블


    // 1. 초기화: Coda에서 학생 명단 불러오기

    async function loadStudents() {

        try {

            const response = await fetch(`https://coda.io/api/v1/docs/${DOC_ID}/tables/${STUDENT_TABLE_ID}/rows`, {

                headers: { "Authorization": `Bearer ${API_TOKEN}` }

            });

            const data = await response.json();

            // 학생 번호를 키로, 이름을 값으로 저장 (컬럼명에 맞춰 수정 필요)

            data.items.forEach(item => {

                const phone = item.values['c-xxxxxxxxxx']; // '번호' 컬럼 ID

                const name = item.values['c-yyyyyyyyyy'];  // '이름' 컬럼 ID

                studentData[phone] = name;

            });

            status.innerText = "준비 완료!";

        } catch (e) {

            status.innerText = "데이터 로드 실패. 인터넷을 확인하세요.";

        }

    }


    // 2. 음성 안내 함수 (TTS)

    function speak(text) {

        const msg = new SpeechSynthesisUtterance(text);

        msg.lang = 'ko-KR';

        msg.rate = 1.0;

        window.speechSynthesis.speak(msg);

    }


    function addNum(num) {

        if (input.length < 4) {

            input += num;

            updateDisplay();

            

            // 4자리가 입력되면 학생 이름 검색

            if (input.length === 4) {

                if (studentData[input]) {

                    studentNameDiv.innerText = `${studentData[input]} 학생 맞나요?`;

                } else {

                    studentNameDiv.innerText = "등록되지 않은 번호입니다.";

                }

            }

        }

    }


    function clearNum() {

        input = "";

        updateDisplay();

        studentNameDiv.innerText = "반갑습니다!";

        status.innerText = "번호를 입력해주세요.";

    }


    function updateDisplay() {

        display.innerText = input.padEnd(4, '-');

    }


    // 3. 출결 전송 및 음성 피드백

    async function submitAttendance() {

        const name = studentData[input];

        if (!name) {

            status.innerText = "번호를 다시 확인해주세요.";

            return;

        }


        status.innerText = "전송 중...";

        spinner.style.display = "block";


        try {

            const response = await fetch(`https://coda.io/api/v1/docs/${DOC_ID}/tables/${LOG_TABLE_ID}/rows`, {

                method: "POST",

                headers: {

                    "Authorization": `Bearer ${API_TOKEN}`,

                    "Content-Type": "application/json"

                },

                body: JSON.stringify({

                    "rows": [{ "cells": [

                        { "column": "학생성함", "value": name },

                        { "column": "시간", "value": new Date().toLocaleString() }

                    ]}]

                })

            });


            if (response.ok) {

                speak(`${name} 학생, 등원 확인되었습니다.`);

                status.innerText = "출결 확인 완료!";

                setTimeout(clearNum, 3000);

            } else { throw new Error(); }

        } catch (e) {

            status.innerText = "오류 발생! 다시 시도하세요.";

        } finally {

            spinner.style.display = "none";

        }

    }


    window.onload = loadStudents;

</script>
